          if (catProducts.length === 0) return null;

          return (
            <div key={cat.id} className="store-catalog-group">
              {/* Category Group Header */}
              <header className="flex items-center justify-between pb-2 mb-3.5 border-b border-cyan-500/15">
                <h2 className="flex items-center gap-2 text-sm sm:text-base font-bold text-white tracking-wide">
                  {/* eslint-disable-next-line @next/next/no-img-element */}
                  <img
                    src="/brand/logo.png"
                    alt="ARX Logo"
                    className="w-4.5 h-4.5 object-contain drop-shadow-[0_0_6px_rgba(0,229,255,0.7)]"
                    onError={(e) => {
                      const target = e.currentTarget;
                      if (!target.src.includes('arx-default-logo.png')) {
                        target.src = '/brand/arx-default-logo.png';
                      }
                    }}
                  />
                  <span className="text-white">{cat.name}</span>
                </h2>
                <span className="text-[11px] text-slate-400 bg-white/5 px-2 py-0.5 rounded border border-white/5">
                  <strong className="text-cyan-300 font-bold">{catProducts.length}</strong> {t('items_count')}
                </span>
              </header>

              {/* Product Cards Grid: 4 columns like original */}
              <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 lg:grid-cols-4 gap-3 sm:gap-3.5">
                {catProducts.map((product) => {
                  const totalStock = product.packages.reduce((sum, p) => sum + (p.stock || 0), 0);
                  const isSoldOut = product.packages.every(pkg => !pkg.inStock || pkg.stock <= 0) || totalStock <= 0 || product.status === 'out_of_stock';
                  const minPrice = Math.min(...product.packages.map(p => p.price));
                  const maxPrice = Math.max(...product.packages.map(p => p.price));

                  return (
                    <div
                      key={product.id}
                      id={`product-card-${product.id}`}
                      data-product-id={product.id}
                      onClick={() => {
                        playClick();
                        router.push(`/products/${product.id}`);
                      }}
                      className={`group cursor-pointer rounded-xl bg-[#0a0d16]/90 border border-cyan-500/15 hover:border-cyan-400/50 p-2.5 transition-all duration-300 hover:-translate-y-1 hover:shadow-xl hover:shadow-cyan-500/15 flex flex-col relative arx-product-card ${
                        previewHoverId === product.id ? 'hovered border-cyan-400/50 -translate-y-1 shadow-xl shadow-cyan-500/15' : ''
                      }`}
                    >
                      {/* Product Thumbnail (1500x1500 Square 1:1 Aspect Ratio) */}
                      <div className="relative aspect-square rounded-lg overflow-hidden bg-black mb-2">
                        {/* eslint-disable-next-line @next/next/no-img-element */}
                        <img
                          src={product.image}
                          alt={product.name}
                          className={`arx-card-thumb w-full h-full object-cover transition-all duration-400 group-hover:scale-105 group-hover:brightness-75 group-hover:blur-[1px] ${
                            isSoldOut ? 'grayscale contrast-125' : ''
                          }`}
                          loading="lazy"
                          onError={(e) => {
                            const target = e.currentTarget;
                            if (!target.src.includes('46ebc6a271269458ec9b.webp')) {
                              target.src = '/catalog/imported/46ebc6a271269458ec9b.webp';
                            }
                          }}
                        />

                        {/* ARX Cyber HUD Product Hover Loader with Shop Logo */}
                        <ProductHoverLoader
                          label={t('details') || 'CLICK TO VIEW'}
                          isSoldOut={isSoldOut}
                        />

                        {/* Sold Out / Out of Stock Overlay Stamp (when not hovered) */}
                        {isSoldOut && (
                          <div className="absolute inset-0 bg-black/65 backdrop-blur-[2px] flex items-center justify-center group-hover:opacity-0 transition-opacity duration-200">
                            <span className="px-2.5 py-0.5 rounded bg-red-600/95 text-white font-black text-[10px] tracking-wider uppercase border border-red-400/60 rotate-[-8deg] shadow-lg">
                              OUT OF STOCK
                            </span>
                          </div>
                        )}

                        {/* Badge if available (hide if sold out or on hover to keep HUD clean) */}
                        {!isSoldOut && product.badge && product.badge !== 'OUT OF STOCK' && product.badge !== 'สินค้าหมด' && (
                          <div className="absolute top-1.5 right-1.5 group-hover:opacity-0 group-[.hovered]:opacity-0 transition-opacity duration-200">
                            <span className="px-1.5 py-0.5 rounded-full text-[9px] font-black backdrop-blur-md shadow-md bg-[#00e5ff] text-slate-950 shadow-cyan-500/30">
                              {product.badge}
                            </span>
                          </div>
                        )}
                      </div>

                      {/* Meta Row: Stock & View details link */}
                      <div className="flex items-center justify-between text-[10px] text-slate-400 mb-1 px-0.5">
                        <span className={isSoldOut ? 'text-red-400 font-bold uppercase tracking-wider' : 'text-slate-400'}>
                          {isSoldOut ? 'OUT OF STOCK' : `${t('in_stock')} · ${totalStock} ${t('items_suffix')}`}
                        </span>
                        <span className="text-slate-400 group-hover:text-[#00e5ff] flex items-center gap-0.5 transition-colors font-medium">
                          {t('details')} <ArrowUpRight className="w-2.5 h-2.5 group-hover:translate-x-0.5 group-hover:-translate-y-0.5 transition-transform" />
                        </span>
                      </div>

                      {/* Product Title */}
                      <h3 className="text-xs sm:text-[13px] font-semibold text-white tracking-wide group-hover:text-cyan-300 transition-colors line-clamp-1 mb-1.5 px-0.5">
                        {product.name}
                      </h3>

                      {/* Price Range and Explicit Buy/View Button */}
                      <div className="mt-auto pt-1.5 px-0.5 flex items-center justify-between border-t border-white/5">
                        <div className="text-xs sm:text-[13px] font-bold text-cyan-400 tracking-tight font-mono">
                          {minPrice === maxPrice ? `฿${minPrice.toLocaleString()}` : `฿${minPrice.toLocaleString()} - ฿${maxPrice.toLocaleString()}`}
                        </div>

                        {isSoldOut ? (
                          <button
                            type="button"
                            onClick={(e) => {
                              e.stopPropagation();
                              playClick();
                              router.push(`/products/${product.id}`);
                            }}
                            className="px-2 py-0.5 rounded-md bg-red-500/15 text-red-400 border border-red-500/30 text-[9px] font-black uppercase tracking-wider flex items-center gap-1 transition-all duration-200 cursor-pointer hover:bg-red-500/25"
                          >
                            OUT OF STOCK
                          </button>
                        ) : (
                          <button
                            type="button"
                            onClick={(e) => {
                              e.stopPropagation();
                              playClick();
                              router.push(`/products/${product.id}`);
                            }}
                            className="px-2 py-0.5 rounded-md bg-cyan-500/15 group-hover:bg-[#00e5ff] text-cyan-300 group-hover:text-slate-950 border border-cyan-500/30 group-hover:border-cyan-400 text-[10px] font-bold flex items-center gap-1 transition-all duration-200 shadow-sm cursor-pointer"
                          >
                            <ShoppingBag className="w-2.5 h-2.5" />
                            <span>{t('buy_now')}</span>
                          </button>
                        )}
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          );
