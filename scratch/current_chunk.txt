          if (catProducts.length === 0) return null;

          const totalCatStock = catProducts.reduce((sum, p) => sum + p.packages.reduce((pSum, pkg) => pSum + (pkg.stock || 0), 0), 0);

          return (
            <div
              key={cat.id}
              className="store-catalog-group rounded-2xl bg-[#070b14]/85 border border-white/[0.08] hover:border-cyan-500/30 p-4 sm:p-5 shadow-xl shadow-black/40 space-y-4 relative overflow-hidden transition-all duration-300"
            >
              {/* Category Ambient Top Glow */}
              <div className="absolute top-0 inset-x-0 h-[2px] bg-gradient-to-r from-transparent via-cyan-500/40 to-transparent pointer-events-none" />

              {/* Category Group Header Bar */}
              <header className="flex flex-col sm:flex-row sm:items-center justify-between gap-2 pb-3 border-b border-white/10">
                <div className="flex items-center gap-2.5">
                  <span className="w-2.5 h-2.5 rounded-full bg-cyan-400 shadow-[0_0_10px_#00e5ff]" />
                  <h2 className="flex items-center gap-2 text-sm sm:text-base font-black text-white tracking-wide uppercase">
                    {/* eslint-disable-next-line @next/next/no-img-element */}
                    <img
                      src="/brand/logo.png"
                      alt="ARX Logo"
                      className="w-4.5 h-4.5 object-contain drop-shadow-[0_0_6px_rgba(0,229,255,0.7)]"
                      onError={(e) => {
                        const target = e.currentTarget;
                        if (!target.src.includes('arx-default-logo.png')) {
                          target.src = '/brand/arx-default-logo.png';
                        }
                      }}
                    />
                    <span>{cat.name}</span>
                    <span className="text-[10px] text-cyan-300 font-bold font-mono bg-cyan-500/15 px-2 py-0.5 rounded-full border border-cyan-500/30">
                      {catProducts.length} สินค้า
                    </span>
                  </h2>
                </div>

                <div className="flex items-center gap-2 text-xs text-slate-400 font-mono">
                  <span>พร้อมส่งในหมวดหมู่นี้:</span>
                  <span className="text-emerald-400 font-bold bg-emerald-500/15 px-2 py-0.5 rounded-md border border-emerald-500/30">
                    {totalCatStock} ชิ้น
                  </span>
                </div>
              </header>

              {/* Product Cards Grid: 1 แถบ มี 2 สินค้า (2 products per row) */}
              <div className="grid grid-cols-1 lg:grid-cols-2 gap-3.5 sm:gap-4">
                {catProducts.map((product) => {
                  const totalStock = product.packages.reduce((sum, p) => sum + (p.stock || 0), 0);
                  const isSoldOut = product.packages.every(pkg => !pkg.inStock || pkg.stock <= 0) || totalStock <= 0 || product.status === 'out_of_stock';
                  const minPrice = Math.min(...product.packages.map(p => p.price));
                  const maxPrice = Math.max(...product.packages.map(p => p.price));

                  return (
                    <div
                      key={product.id}
                      id={`product-card-${product.id}`}
                      data-product-id={product.id}
                      onClick={() => {
                        playClick();
                        router.push(`/products/${product.id}`);
                      }}
                      className={`group cursor-pointer rounded-xl bg-[#090e1a]/90 hover:bg-[#0d1424] border border-cyan-500/20 hover:border-cyan-400/60 p-3 sm:p-3.5 transition-all duration-300 hover:-translate-y-1 hover:shadow-xl hover:shadow-cyan-500/15 flex flex-col sm:flex-row gap-3.5 relative overflow-hidden arx-product-card ${
                        previewHoverId === product.id ? 'hovered border-cyan-400/60 -translate-y-1 shadow-xl shadow-cyan-500/15' : ''
                      }`}
                    >
                      {/* Ambient Card Top Glow */}
                      <div className="absolute top-0 inset-x-0 h-[1px] bg-gradient-to-r from-transparent via-cyan-500/30 group-hover:via-cyan-400/80 to-transparent pointer-events-none transition-all" />

                      {/* Product Thumbnail (Square 1:1 Aspect Ratio) */}
                      <div className="relative w-full sm:w-36 md:w-40 sm:h-36 md:h-40 aspect-square sm:aspect-auto rounded-lg overflow-hidden bg-black shrink-0 border border-white/10 group-hover:border-cyan-500/40 transition-colors">
                        {/* eslint-disable-next-line @next/next/no-img-element */}
                        <img
                          src={product.image}
                          alt={product.name}
                          className={`arx-card-thumb w-full h-full object-cover transition-all duration-400 group-hover:scale-105 group-hover:brightness-75 group-hover:blur-[1px] ${
                            isSoldOut ? 'grayscale contrast-125' : ''
                          }`}
                          loading="lazy"
                          onError={(e) => {
                            const target = e.currentTarget;
                            if (!target.src.includes('46ebc6a271269458ec9b.webp')) {
                              target.src = '/catalog/imported/46ebc6a271269458ec9b.webp';
                            }
                          }}
                        />

                        {/* ARX Cyber HUD Product Hover Loader */}
                        <ProductHoverLoader
                          label={t('details') || 'CLICK TO VIEW'}
                          isSoldOut={isSoldOut}
                        />

                        {/* Sold Out / Out of Stock Overlay Stamp */}
                        {isSoldOut && (
                          <div className="absolute inset-0 bg-black/70 backdrop-blur-[2px] flex items-center justify-center group-hover:opacity-0 transition-opacity duration-200">
                            <span className="px-2.5 py-0.5 rounded bg-red-600 text-white font-black text-[10px] tracking-wider uppercase border border-red-400/60 rotate-[-8deg] shadow-lg">
                              OUT OF STOCK
                            </span>
                          </div>
                        )}

                        {/* Badge if available */}
                        {!isSoldOut && product.badge && product.badge !== 'OUT OF STOCK' && product.badge !== 'สินค้าหมด' && (
                          <div className="absolute top-1.5 right-1.5 group-hover:opacity-0 group-[.hovered]:opacity-0 transition-opacity duration-200">
                            <span className="px-1.5 py-0.5 rounded-full text-[9px] font-black backdrop-blur-md shadow-md bg-[#00e5ff] text-slate-950 shadow-cyan-500/30">
                              {product.badge}
                            </span>
                          </div>
                        )}
                      </div>

                      {/* Product Information Column */}
                      <div className="flex-1 min-w-0 flex flex-col justify-between">
                        <div>
                          {/* Badges / Stock row */}
                          <div className="flex items-center justify-between gap-1.5 mb-1">
                            <span className="text-[9.5px] px-2 py-0.2 rounded-full font-bold bg-cyan-500/15 text-cyan-300 border border-cyan-500/30 truncate max-w-[130px]">
                              {cat.name}
                            </span>
                            <span className={`text-[10px] font-mono font-bold ${isSoldOut ? 'text-red-400' : 'text-emerald-400'}`}>
                              {isSoldOut ? 'สินค้าหมด' : `พร้อมส่ง ${totalStock} ชิ้น`}
                            </span>
                          </div>

                          {/* Product Title */}
                          <h3 className="text-sm sm:text-[15px] font-bold text-white tracking-wide group-hover:text-cyan-300 transition-colors line-clamp-1 mb-1">
                            {product.name}
                          </h3>

                          {/* Short Description */}
                          <p className="text-[11px] text-slate-400 line-clamp-2 leading-relaxed mb-2 font-light">
                            {product.shortDescription || 'โปรแกรมคุณภาพสูง ระบบเสถียร รองรับ Windows 10/11 อัปเดตอัตโนมัติ'}
                          </p>

                          {/* Available Packages pills */}
                          <div className="flex items-center gap-1.5 flex-wrap mb-2">
                            {product.packages.slice(0, 3).map(pkg => (
                              <span key={pkg.id} className="text-[9px] px-1.5 py-0.2 rounded bg-slate-900 border border-white/10 text-slate-300 font-mono">
                                {pkg.duration} <strong className="text-cyan-400 font-bold">฿{pkg.price}</strong>
                              </span>
                            ))}
                            {product.packages.length > 3 && (
                              <span className="text-[9px] text-slate-500 font-mono">+{product.packages.length - 3}</span>
                            )}
                          </div>
                        </div>

                        {/* Bottom Row: Price and Action Button */}
                        <div className="pt-2 border-t border-white/5 flex items-center justify-between mt-auto">
                          <div>
                            <span className="text-[9px] text-slate-400 block leading-none mb-0.5">ราคาเริ่มต้น</span>
                            <div className="text-sm sm:text-base font-bold text-cyan-400 tracking-tight font-mono">
                              {minPrice === maxPrice ? `฿${minPrice.toLocaleString()}` : `฿${minPrice.toLocaleString()} - ฿${maxPrice.toLocaleString()}`}
                            </div>
                          </div>

                          {isSoldOut ? (
                            <button
                              type="button"
                              onClick={(e) => {
                                e.stopPropagation();
                                playClick();
                                router.push(`/products/${product.id}`);
                              }}
                              className="px-2.5 py-1 rounded-md bg-red-500/15 text-red-400 border border-red-500/30 text-[9.5px] font-black uppercase tracking-wider flex items-center gap-1 cursor-pointer"
                            >
                              สินค้าหมด
                            </button>
                          ) : (
                            <button
                              type="button"
                              onClick={(e) => {
                                e.stopPropagation();
                                playClick();
                                router.push(`/products/${product.id}`);
                              }}
                              className="px-3 py-1 rounded-md bg-gradient-to-r from-cyan-500 to-blue-600 hover:from-cyan-400 hover:to-blue-500 text-slate-950 font-bold text-[10.5px] flex items-center gap-1 transition-all shadow-sm shadow-cyan-500/20 cursor-pointer"
                            >
                              <ShoppingBag className="w-3 h-3" />
                              <span>สั่งซื้อทันที</span>
                            </button>
                          )}
                        </div>
                      </div>
                    </div>
                  );
                })}
              </div>
            </div>
          );
